cmake_minimum_required(VERSION 3.20)
project(cpu_gpu_math_lab VERSION 0.1.0 LANGUAGES CXX)
option(MATHLAB_NATIVE "Tune for this machine (binaries may not run on another CPU)" OFF)

add_library(mathlab STATIC src/division.cpp)
target_include_directories(mathlab PUBLIC include)
target_compile_features(mathlab PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(mathlab PRIVATE /O2 /fp:precise)
  if(MATHLAB_NATIVE)
    target_compile_options(mathlab PRIVATE /arch:AVX2)
  endif()
else()
  target_compile_options(mathlab PRIVATE -O3 -fno-math-errno -ffp-contract=off)
  if(MATHLAB_NATIVE)
    target_compile_options(mathlab PRIVATE -march=native)
  endif()
endif()
target_compile_definitions(mathlab PUBLIC MATHLAB_NATIVE_BUILD=$<BOOL:${MATHLAB_NATIVE}>)

add_executable(mathlab_bench src/benchmark.cpp)
target_link_libraries(mathlab_bench PRIVATE mathlab)
add_executable(mathlab_tests tests/division_test.cpp)
target_link_libraries(mathlab_tests PRIVATE mathlab)
enable_testing()
add_test(NAME exact_uint32_division COMMAND mathlab_tests)
